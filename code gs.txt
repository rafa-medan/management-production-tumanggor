function catatLog(username, role, aktivitas, keterangan) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    if (!sheetLog) { setupSheets(); sheetLog = ss.getSheetByName("LogAktivitas"); }

    sheetLog.appendRow([waktu, username || "Public/Guest", role || "GUEST", aktivitas, keterangan || ""]);
  } catch (e) {
    console.error("Gagal log: " + e.toString());
  }
}

// 2. GET PUBLIC DATA UNTUK FRONTEND
function getPublicPortalData() {
  setupSheets();
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // Settings
  var settings = {};
  var sData = ss.getSheetByName("Settings").getDataRange().getValues();
  for (var i = 1; i < sData.length; i++) {
    settings[sData[i][0]] = sData[i][1];
  }

  // Videos
  var videos = [];
  var vData = ss.getSheetByName("Videos").getDataRange().getValues();
  for (var v = 1; v < vData.length; v++) {
    videos.push({
      id: vData[v][0],
      waktu: vData[v][1],
      divisi: vData[v][2],
      kategori: vData[v][3],
      judul: vData[v][4],
      link: vData[v][5]
    });
  }

  // Products
  var products = [];
  var pData = ss.getSheetByName("Products").getDataRange().getValues();
  for (var p = 1; p < pData.length; p++) {
    if (pData[p][5] === "Aktif") {
      products.push({
        id: pData[p][0],
        divisi: pData[p][1],
        nama: pData[p][2],
        harga: pData[p][3],
        deskripsi: pData[p][4]
      });
    }
  }

  // Sales Data untuk Grafik
  var sales = [];
  var slData = ss.getSheetByName("SalesData").getDataRange().getValues();
  for (var s = 1; s < slData.length; s++) {
    sales.push({
      tanggal: Utilities.formatDate(new Date(slData[s][0]), "GMT+7", "yyyy-MM-dd"),
      divisi: slData[s][1],
      total: Number(slData[s][2]) || 0
    });
  }

  return {
    success: true,
    settings: settings,
    videos: videos,
    products: products,
    sales: sales
  };
}

// 3. FITUR VIDEO DIVISI & TESTIMONI (Poin 1)
function uploadVideoDivisi(token, fileData, fileName, mimeType, divisi, kategori, judul) {
  var user = validateToken(token);
  if (!user || (user.role !== "SUPER_ADMIN" && user.role !== "ADMIN_DIVISI")) {
    return { success: false, message: "Akses ditolak! Hanya Admin yang dapat mengunggah video." };
  }

  try {
    var folder = DriveApp.getRootFolder();
    var decodedBlob = Utilities.newBlob(Utilities.base64Decode(fileData), mimeType, fileName);
    var file = folder.createFile(decodedBlob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

    var videoUrl = file.getUrl();
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Videos");
    var id = "VID-" + new Date().getTime();
    var waktu = Utilities.formatDate(new Date(), "GMT+7", "yyyy-MM-dd HH:mm:ss");

    sheet.appendRow([id, waktu, divisi, kategori, judul, videoUrl, user.username]);
    catatLog(user.username, user.role, "Upload Video", "Divisi: " + divisi + ", Kategori: " + kategori);

    return { success: true, message: "Video " + kategori + " berhasil diunggah!" };
  } catch(e) {
    return { success: false, message: "Gagal upload video: " + e.toString() };
  }
}

// 4. FITUR SOSIAL MEDIA & CONTACT PERSON (Poin 2 & 3)
function updateSettingSocialAndContact(token, keyName, valueData) {
  var user = validateToken(token);
  if (!user) return { success: false, message: "Sesi berakhir." };
  
  if (user.role !== "SUPER_ADMIN" && user.role !== "ADMIN_DIVISI") {
    return { success: false, message: "Akses ditolak." };
  }

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Settings");
  var data = sheet.getDataRange().getValues();

  var strValue = (typeof valueData === 'object') ? JSON.stringify(valueData) : String(valueData);
  var updated = false;

  for (var i = 1; i < data.length; i++) {
    if (data[i][0] === keyName) {
      sheet.getRange(i + 1, 2).setValue(strValue);
      updated = true;
      break;
    }
  }

  if (!updated) {
    sheet.appendRow([keyName, strValue]);
  }

  catatLog(user.username, user.role, "Update Setting", keyName);
  return { success: true, message: "Pengaturan berhasil diperbarui!" };
}

// 5. FITUR PRODUK & HARGA (Poin 5)
function saveProduct(token, productObj) {
  var user = validateToken(token);
  if (!user || (user.role !== "SUPER_ADMIN" && user.role !== "ADMIN_DIVISI")) {
    return { success: false, message: "Akses ditolak!" };
  }

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Products");

  if (productObj.id) {
    var data = sheet.getDataRange().getValues();
    for (var i = 1; i < data.length; i++) {
      if (data[i][0] === productObj.id) {
        sheet.getRange(i + 1, 3).setValue(productObj.nama);
        sheet.getRange(i + 1, 4).setValue(productObj.harga);
        sheet.getRange(i + 1, 5).setValue(productObj.deskripsi);
        catatLog(user.username, user.role, "Edit Produk", productObj.nama);
        return { success: true, message: "Produk berhasil diperbarui!" };
      }
    }
  } else {
    var newId = "PRD-" + new Date().getTime();
    sheet.appendRow([newId, productObj.divisi, productObj.nama, productObj.harga, productObj.deskripsi, "Aktif"]);
    catatLog(user.username, user.role, "Tambah Produk", productObj.nama);
    return { success: true, message: "Produk baru berhasil ditambahkan!" };
  }

  return { success: false, message: "Gagal memproses produk." };
}

// 6. FITUR LIVE CHAT 24 JAM & AUTO TRANSLATE (Poin 4)
function sendLiveChatMessage(nama, noHP, pesan, senderRole, targetLang) {
  setupSheets();
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("LiveChats");
  var id = "CHAT-" + new Date().getTime();
  var waktu = Utilities.formatDate(new Date(), "GMT+7", "yyyy-MM-dd HH:mm:ss");

  var finalPesan = pesan;
  if (targetLang && targetLang !== "id") {
    try {
      finalPesan = LanguageApp.translate(pesan, '', targetLang);
    } catch(e){}
  }

  sheet.appendRow([id, waktu, nama, "'" + noHP, senderRole || "CUSTOMER", finalPesan, targetLang || "id"]);
  return { success: true, message: "Pesan terkirim!", translatedText: finalPesan };
}

function registerClientOrPartner(data) {
  if (data.role !== "PARTNER" && data.role !== "CUSTOMER") {
    return { success: false, message: "Role registrasi tidak valid." };
  }

  setupSheets();
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("ClientsPartners");
  var rows = sheet.getDataRange().getValues();

  var cleanUsername = String(data.username).trim();
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][0]).trim().toLowerCase() === cleanUsername.toLowerCase()) {
      return { success: false, message: "Username '" + cleanUsername + "' sudah terdaftar!" };
    }
  }

  var kodeVerifikasi = Math.floor(1000 + Math.random() * 9000).toString();

  sheet.appendRow([
    cleanUsername,
    data.password,
    data.nama,
    data.role,
    "'" + data.telepon,
    data.email,
    data.alamat || "-",
    data.institusi || "-",
    data.jabatan || "-",
    kodeVerifikasi,
    "PENDING"
  ]);
  
  catatLog(cleanUsername, data.role, "Registrasi Mandiri Klien/Partner", "Nama: " + data.nama);

  try {
    var htmlBody = `
      <div style="font-family: Arial, sans-serif; background-color: #0a0a0b; color: #ffffff; padding: 25px; border-radius: 10px;">
        <h2 style="color: #d4af37; border-bottom: 1px solid #d4af37; padding-bottom: 10px;">Verifikasi Akun MultiCorp (${data.role})</h2>
        <p>Halo <strong>${data.nama}</strong>,</p>
        <p>Terima kasih telah mendaftar di Portal MultiCorp Enterprise.</p>
        <div style="background: #141416; padding: 15px; text-align: center; border: 1px dashed #d4af37; margin: 20px 0;">
          <small style="color: #aaa;">KODE VERIFIKASI EMAIL ANDA:</small>
          <h1 style="color: #d4af37; letter-spacing: 5px; margin: 10px 0;">${kodeVerifikasi}</h1>
        </div>
        <p>Masukkan 4 digit kode di atas pada portal untuk mengaktifkan akun Anda.</p>
      </div>
    `;
    GmailApp.sendEmail(data.email, "Kode Verifikasi Akun MultiCorp Portal", "Kode Verifikasi: " + kodeVerifikasi, { htmlBody: htmlBody });
  } catch (e) {
    return { success: false, message: "Gagal mengirim email verifikasi: " + e.toString() };
  }

  return { 
    success: true, 
    requiresVerification: true, 
    username: cleanUsername,
    message: "Registrasi berhasil! Kode verifikasi 4-digit telah dikirimkan ke email Anda." 
  };
}

function validateToken(token) {
  try {
    var decoded = Utilities.newBlob(Utilities.base64Decode(token)).getDataAsString();
    var sessionObj = JSON.parse(decoded);
    if (new Date().getTime() > sessionObj.expire) return null;
    return sessionObj;
  } catch (e) {
    return null;
  }
}

function getAdminDashboardData(token) {
  var user = validateToken(token);
  if (!user) return { success: false, message: "Sesi telah berakhir. Silakan login kembali." };

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  var settings = {};
  var sheetSettings = ss.getSheetByName("Settings");
  if (sheetSettings) {
    var sData = sheetSettings.getDataRange().getValues();
    for(var s=1; s<sData.length; s++) {
      settings[sData[s][0]] = sData[s][1];
    }
  }

  var messages = [];
  if (user.role === "SUPER_ADMIN" || user.role === "ADMIN_DIVISI") {
    var sheetPesan = ss.getSheetByName("Pesan");
    var dataPesan = sheetPesan ? sheetPesan.getDataRange().getValues() : [];
    for (var i = 1; i < dataPesan.length; i++) {
      var row = dataPesan[i];
      if (user.role === "SUPER_ADMIN" || user.divisi === row[2]) {
        messages.push({
          id: row[0],
          timestamp: Utilities.formatDate(new Date(row[1]), "GMT+7", "yyyy-MM-dd HH:mm:ss"),
          divisi: row[2],
          nama: row[3],
          email: row[4],
          telepon: String(row[5]).replace("'", ""),
          pesan: row[6],
          status: row[7] || "Pending"
        });
      }
    }
  }

  var users = [];
  var clientsPartners = [];
  if (user.role === "SUPER_ADMIN" || user.role === "ADMIN_DIVISI") {
    var sheetUsers = ss.getSheetByName("Users");
    var dataUsers = sheetUsers ? sheetUsers.getDataRange().getValues() : [];
    for (var j = 1; j < dataUsers.length; j++) {
      var uRow = dataUsers[j];
      if (user.role === "SUPER_ADMIN" || uRow[4] === user.divisi) {
        users.push({
          username: uRow[0],
          password: uRow[1],
          nama: uRow[2],
          role: uRow[3],
          divisi: uRow[4],
          telepon: String(uRow[5] || "").replace("'", ""),
          email: uRow[6],
          alamat: uRow[7] || "-",
          institusi: uRow[8] || "-",
          jabatan: uRow[9] || "-",
          statusVerifikasi: uRow[11] || "VERIFIED"
        });
      }
    }

    if (user.role === "SUPER_ADMIN") {
      var sheetClients = ss.getSheetByName("ClientsPartners");
      var dataClients = sheetClients ? sheetClients.getDataRange().getValues() : [];
      for (var c = 1; c < dataClients.length; c++) {
        var cRow = dataClients[c];
        clientsPartners.push({
          username: cRow[0],
          password: cRow[1],
          nama: cRow[2],
          role: cRow[3],
          telepon: String(cRow[4] || "").replace("'", ""),
          email: cRow[5],
          alamat: cRow[6] || "-",
          institusi: cRow[7] || "-",
          jabatan: cRow[8] || "-",
          statusVerifikasi: cRow[10] || "VERIFIED"
        });
      }
    }
  }

  var files = [];
  var sheetFiles = ss.getSheetByName("BerkasUpload");
  var dataFiles = sheetFiles ? sheetFiles.getDataRange().getValues() : [];
  for (var k = 1; k < dataFiles.length; k++) {
    var fRow = dataFiles[k];
    if (user.role === "SUPER_ADMIN" || user.role === "ADMIN_DIVISI" || user.role === "STAFF" || fRow[2].indexOf(user.username) !== -1) {
      files.push({
        id: fRow[0],
        waktu: fRow[1],
        uploader: fRow[2],
        role: fRow[3],
        namaFile: fRow[4],
        link: fRow[5],
        ket: fRow[6]
      });
    }
  }

  var payments = [];
  var sheetPayments = ss.getSheetByName("BuktiPembayaran");
  var dataPayments = sheetPayments ? sheetPayments.getDataRange().getValues() : [];
  for (var p = 1; p < dataPayments.length; p++) {
    var pRow = dataPayments[p];
    if (user.role === "SUPER_ADMIN" || user.role === "ADMIN_DIVISI" || pRow[2].indexOf(user.username) !== -1) {

        waktu: pRow[1],
        uploader: pRow[2],
        role: pRow[3],
        jumlah: pRow[4],
        metode: pRow[5],
        namaFile: pRow[6],
        link: pRow[7],
        ket: pRow[8]
      });
    }
  }

  return { 
    success: true, 
    messages: messages.reverse(), 
    users: users, 
    clientsPartners: clientsPartners,
    files: files.reverse(),
    payments: payments.reverse(),
    settings: settings 
  };
}

function uploadBuktiPembayaran(token, fileData, fileName, mimeType, jumlah, metode, keterangan) {
  var user = validateToken(token);
  if (!user) return { success: false, message: "Sesi telah berakhir. Silakan login ulang." };

  try {
    var folder = DriveApp.getRootFolder();
    var decodedBlob = Utilities.newBlob(Utilities.base64Decode(fileData), mimeType, fileName);
    var file = folder.createFile(decodedBlob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    setupSheets();
    var sheetPayments = ss.getSheetByName("BuktiPembayaran");

    var idBayar = "PAY-" + new Date().getTime();
    var waktu = Utilities.formatDate(new Date(), "GMT+7", "yyyy-MM-dd HH:mm:ss");

    sheetPayments.appendRow([
      idBayar,
      waktu,
      user.nama + " (" + user.username + ")",
      user.role,
      jumlah,
      metode,
      fileName,
      file.getUrl(),
      keterangan || "-"
    ]);

    catatLog(user.username, user.role, "Upload Bukti Pembayaran", "ID: " + idBayar + ", Rp " + jumlah);
    return { success: true, message: "Bukti pembayaran berhasil diunggah ke Administrator!" };
  } catch (err) {
    return { success: false, message: "Gagal mengunggah bukti pembayaran: " + err.toString() };
  }
}

function updateNomorRekening(token, rekeningInfo) {

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    setupSheets();
    var sheetSettings = ss.getSheetByName("Settings");
    var data = sheetSettings.getDataRange().getValues();

    var updated = false;
    for (var i = 1; i < data.length; i++) {
      if (data[i][0] === "REKENING_INFO") {
        sheetSettings.getRange(i + 1, 2).setValue(rekeningInfo);
        updated = true;

    }

    if (bgData && bgName) {
      var bgBlob = Utilities.newBlob(Utilities.base64Decode(bgData), "image/jpeg", bgName);
      var bgFile = folder.createFile(bgBlob);
      bgFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      bgUrl = "https://lh3.googleusercontent.com/d/" + bgFile.getId();
      
      var dataS2 = sheetSettings.getDataRange().getValues();
      var foundBg = false;
      for (var j = 1; j < dataS2.length; j++) {
        if (dataS2[j][0] === "BG_URL") {
          sheetSettings.getRange(j + 1, 2).setValue(bgUrl);
          foundBg = true; break;
        }
      }
      if (!foundBg) sheetSettings.appendRow(["BG_URL", bgUrl]);
    }

    catatLog(user.username, user.role, "Update Branding Utama", "Logo / Background website diubah");
    return { success: true, message: "Branding Logo & Background website berhasil diperbarui!", logoUrl: logoUrl, bgUrl: bgUrl };
  } catch (e) {
    return { success: false, message: "Gagal menyimpan branding: " + e.toString() };
  }
}

function updateStatusPesan(token, idPesan, statusBaru) {
  var user = validateToken(token);

  for (var i = 1; i < data.length; i++) {
    if (data[i][0] === idPesan) {
      sheet.getRange(i + 1, 8).setValue(statusBaru);
      
      var namaKlien = data[i][3];
      var emailKlien = data[i][4];
      var divisiPesan = data[i][2];
      
      try {
        var statusColor = statusBaru === 'Selesai' ? '#28a745' : (statusBaru === 'Proses' ? '#ffc107' : '#dc3545');
        var htmlBody = `
          <div style="font-family: Arial, sans-serif; background-color: #0a0a0b; color: #ffffff; padding: 25px; border-radius: 10px;">
            <h2 style="color: #d4af37; border-bottom: 1px solid #d4af37; padding-bottom: 10px;">Update Status Pengajuan</h2>
            <p>Halo <strong>${namaKlien}</strong>,</p>
            <p>Status pengajuan layanan Anda (${divisiPesan} - ID: ${idPesan}) telah diperbarui:</p>
            <div style="background: #141416; padding: 15px; border-left: 4px solid #d4af37; margin: 20px 0;">
              <p style="margin: 0;">Status Terbaru: <span style="color: ${statusColor}; font-weight: bold; font-size: 16px;">${statusBaru}</span></p>
            </div>
          </div>
        `;

      return { success: true, message: "Status diperbarui dan email pemberitahuan terkirim!" };

  }
  return { success: false, message: "Pesan tidak ditemukan." };
}

function updateAdminCredentials(token, targetUsername, passwordBaru, teleponBaru) {
  var user = validateToken(token);
  if (!user) return { success: false, message: "Sesi berakhir." };
  
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Users");
  var data = sheet.getDataRange().getValues();
  
  for (var i = 1; i < data.length; i++) {
    var rowUsername = String(data[i][0]).trim();
    if (rowUsername.toLowerCase() === String(targetUsername).toLowerCase()) {
      if (passwordBaru) sheet.getRange(i + 1, 2).setValue(passwordBaru);
      if (teleponBaru !== undefined) sheet.getRange(i + 1, 6).setValue("'" + teleponBaru);
      catatLog(user.username, user.role, "Update Kredensial", "Target: " + targetUsername);
      return { success: true, message: "Kredensial berhasil diperbarui!" };
    }
  }

  var sheetC = ss.getSheetByName("ClientsPartners");
  var dataC = sheetC.getDataRange().getValues();
  for (var j = 1; j < dataC.length; j++) {
    if (String(dataC[j][0]).toLowerCase() === String(targetUsername).toLowerCase()) {
      if (passwordBaru) sheetC.getRange(j + 1, 2).setValue(passwordBaru);
      if (teleponBaru !== undefined) sheetC.getRange(j + 1, 5).setValue("'" + teleponBaru);
      catatLog(user.username, user.role, "Update Kredensial Client/Partner", "Target: " + targetUsername);
      return { success: true, message: "Kredensial Partner/Customer berhasil diperbarui!" };
    }
  }

  return { success: false, message: "User tidak ditemukan." };
}

function hapusAdmin(token, usernameTarget) {
  var user = validateToken(token);
  if (!user || user.role !== "SUPER_ADMIN") return { success: false, message: "Akses ditolak! Hanya Super Admin yang dapat menghapus akun." };
  
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Users");
  var data = sheet.getDataRange().getValues();
  
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]).toLowerCase() === String(usernameTarget).toLowerCase()) {
      sheet.deleteRow(i + 1);
      catatLog(user.username, user.role, "Hapus User", "Target: " + usernameTarget);
      return { success: true, message: "Akun berhasil dihapus!" };
    }
  }

  var sheetC = ss.getSheetByName("ClientsPartners");
  var dataC = sheetC.getDataRange().getValues();
  for (var j = 1; j < dataC.length; j++) {
    if (String(dataC[j][0]).toLowerCase() === String(usernameTarget).toLowerCase()) {
      sheetC.deleteRow(j + 1);
      catatLog(user.username, user.role, "Hapus Client/Partner", "Target: " + usernameTarget);
      return { success: true, message: "Akun Partner/Customer berhasil dihapus!" };
    }
  }

  return { success: false, message: "Target tidak ditemukan." };
}
