    function loadPublicPortalData() {
      google.script.run.withSuccessHandler(function(res) {
        if (!res || !res.success) return;
        publicDataCache = res;

        // Render Contact Person (Poin 2)
        if (res.settings.CONTACT_PERSON) {
          document.getElementById('publicContactBox').innerText = res.settings.CONTACT_PERSON;

        // Render Sosmed Publik (Poin 3)
        if (res.settings.SOCMED_PUBLIC) {
          renderSocmedIcons(JSON.parse(res.settings.SOCMED_PUBLIC), 'publicSocmedIcons');
        }

        // Render Company Profile (Poin 7)
        if (res.settings.COMPANY_PROFILE) {
          let cp = JSON.parse(res.settings.COMPANY_PROFILE);
          document.getElementById('cpTahunBerdiri').innerText = cp.tahunBerdiri || '2018';
          document.getElementById('cpProfileText').innerText = cp.profile || '';
          document.getElementById('cpVisiText').innerText = cp.visi || '';
          document.getElementById('cpMisiText').innerText = cp.misi || '';
          document.getElementById('cpTujuanText').innerText = cp.tujuan || '';
        }

        // Render Developer Info (Poin 9)
        if (res.settings.WEB_DEVELOPER_INFO) {
          let dev = JSON.parse(res.settings.WEB_DEVELOPER_INFO);
          document.getElementById('webDevInfo').innerText = `Website dikembangkan oleh: ${dev.nama} (${dev.tahun})`;
        }

        // Render Cara Order (Poin 10)
        if (res.settings.CARA_ORDER) {
          let list = JSON.parse(res.settings.CARA_ORDER);
          let html = '';
          list.forEach(item => html += `<li class="mb-1">${item}</li>`);
          document.getElementById('caraOrderList').innerHTML = html;
        }

        // Render Logos
        if (res.settings.LOGO_URL) {
          document.getElementById('brandLogoContainer').innerHTML = `<img src="${res.settings.LOGO_URL}" class="navbar-brand-img me-2">`;
        }
        if (res.settings.BG_URL) {
          document.body.style.backgroundImage = `url('${res.settings.BG_URL}')`;

        // Render Products (Poin 5)
        renderPublicProducts(res.products);

        // Render Videos (Poin 1)
        renderPublicVideos(res.videos);

        // Render Grafik Sales (Poin 8)
        renderSalesChart();

      }).getPublicPortalData();

    // RENDER GRAFIK PENJUALAN (POIN 8)
    function renderSalesChart() {
      if (!publicDataCache || !publicDataCache.sales) return;
      let ctx = document.getElementById('salesChartCanvas').getContext('2d');
      let period = document.getElementById('salesFilterPeriod').value;

      let sales = publicDataCache.sales;
      let divisions = ['IT Program', 'Network', 'Nusantara Food'];
      let totals = [0, 0, 0];

      divisions.forEach((div, idx) => {
        let sum = sales.filter(s => s.divisi === div).reduce((acc, curr) => acc + curr.total, 0);
        totals[idx] = sum;
      });

      if (salesChartInstance) salesChartInstance.destroy();

      salesChartInstance = new Chart(ctx, {
        type: 'bar',
        data: {
          labels: divisions,
          datasets: [{
            label: 'Pencapaian Penjualan (Rp)',
            data: totals,
            backgroundColor: ['#d4af37', '#e5c158', '#f3e5ab'],
            borderColor: '#b8901c',
            borderWidth: 1
          }]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          plugins: { legend: { labels: { color: '#ffffff' } } },
          scales: {
            x: { ticks: { color: '#f3e5ab' } },
            y: { ticks: { color: '#ffffff' } }
          }
        }
      });
    }

      let done = messages.filter(m => m.status === 'Selesai').length;

      document.getElementById('statDoneCount').innerText = done;

            <td>
              <select class="form-select form-select-sm" onchange="handleUpdateStatusPesan('${m.id}', this.value)">

                <option value="Proses" ${m.status==='Proses'?'selected':''}>Proses</option>

    function handleUploadPayment(e) {

    function handleSaveBranding(e) {
      e.preventDefault();
      let logoInput = document.getElementById('upLogoFile');
      let bgInput = document.getElementById('upBgFile');

      let logoBase64 = "", logoName = "", bgBase64 = "", bgName = "";

      showLoading("Menyimpan branding...");

      let processSave = function() {
        google.script.run.withSuccessHandler(function(res) {
          hideLoading();
          alert(res.message);
          if (res.success) { loadPublicPortalData(); }
        }).saveBrandingSettings(currentToken, logoBase64, logoName, bgBase64, bgName);
      };

      if (logoInput.files.length > 0) {
        let r1 = new FileReader();
        r1.onload = function(evt) {
          logoBase64 = evt.target.result.split(',')[1];
          logoName = logoInput.files[0].name;
          if (bgInput.files.length > 0) {
            let r2 = new FileReader();
            r2.onload = function(evt2) {
              bgBase64 = evt2.target.result.split(',')[1];
              bgName = bgInput.files[0].name;
              processSave();
            };
            r2.readAsDataURL(bgInput.files[0]);
          } else { processSave(); }
        };
        r1.readAsDataURL(logoInput.files[0]);
      } else if (bgInput.files.length > 0) {
        let r2 = new FileReader();
        r2.onload = function(evt2) {
          bgBase64 = evt2.target.result.split(',')[1];
          bgName = bgInput.files[0].name;
          processSave();
        };
        r2.readAsDataURL(bgInput.files[0]);
      }
    }

    function enablePasswordToggles() {
      document.querySelectorAll('input[type="password"]').forEach(function(input) {
        if (input.dataset.passwordToggleReady) return;
        input.dataset.passwordToggleReady = 'true';
        var wrapper = input.parentElement;
        wrapper.style.position = 'relative';
        input.style.paddingRight = '42px';
        var button = document.createElement('button');
        button.type = 'button';
        button.className = 'btn btn-link position-absolute top-50 end-0 translate-middle-y text-gold';
        button.style.textDecoration = 'none';
        button.setAttribute('aria-label', 'Tampilkan password');
        button.innerHTML = '<i class="fa-solid fa-eye"></i>';
        button.addEventListener('click', function() {
          var visible = input.type === 'password';
          input.type = visible ? 'text' : 'password';
          button.setAttribute('aria-label', visible ? 'Sembunyikan password' : 'Tampilkan password');
          button.innerHTML = visible ? '<i class="fa-solid fa-eye-slash"></i>' : '<i class="fa-solid fa-eye"></i>';
        });
        wrapper.appendChild(button);
      });
    }

    enablePasswordToggles();
